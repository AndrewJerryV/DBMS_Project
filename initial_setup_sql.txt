DROP TABLE IF EXISTS schedules;
DROP TABLE IF EXISTS bookings;
DROP TABLE IF EXISTS passengers;
DROP TABLE IF EXISTS staff;
DROP TABLE IF EXISTS routes;
DROP TABLE IF EXISTS buses;
DROP TABLE IF EXISTS drivers;
DROP TABLE IF EXISTS bus_stops;

-- 1. Bus Stops Table
CREATE TABLE IF NOT EXISTS bus_stops (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    pincode VARCHAR(10),
    district VARCHAR(100)
);

-- 2. Drivers Table
CREATE TABLE IF NOT EXISTS drivers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    license_number VARCHAR(50) UNIQUE,
    contact_number VARCHAR(20),
    address VARCHAR(255),
    availability ENUM('available', 'on_leave', 'on_trip') DEFAULT 'available'
);

-- 3. Buses Table
CREATE TABLE IF NOT EXISTS buses (
    id INT AUTO_INCREMENT PRIMARY KEY,
    bus_number VARCHAR(20) NOT NULL,
    capacity INT NOT NULL,
    driver_id INT,
    bus_stop_id INT, -- where the bus resides when not in use
    status ENUM('active', 'maintenance') DEFAULT 'active',
    bus_type ENUM('AC', 'Non-AC', 'Sleeper', 'Seater') DEFAULT 'Seater',
    FOREIGN KEY (driver_id) REFERENCES drivers(id) ON DELETE SET NULL,
    FOREIGN KEY (bus_stop_id) REFERENCES bus_stops(id)
);

-- 4. Routes Table
CREATE TABLE IF NOT EXISTS routes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    origin_bus_stop_id INT NOT NULL,
    destination_bus_stop_id INT NOT NULL,
    distance_km DECIMAL(6,2) NOT NULL,
    duration_min INT,
    fare DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (origin_bus_stop_id) REFERENCES bus_stops(id),
    FOREIGN KEY (destination_bus_stop_id) REFERENCES bus_stops(id)
);

-- 5. Staff Table
CREATE TABLE IF NOT EXISTS staff (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    role ENUM('admin', 'operator', 'conductor') DEFAULT 'operator',
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(20),
    password VARCHAR(255) NOT NULL, -- store hashed passwords
    salary DECIMAL(10,2),
    date_joined DATE NOT NULL,
    status ENUM('active', 'inactive') DEFAULT 'active'
);

-- 6. Passengers Table
CREATE TABLE IF NOT EXISTS passengers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(20),
    address VARCHAR(255)
);

-- 7. Bookings Table
CREATE TABLE IF NOT EXISTS bookings (
    id INT AUTO_INCREMENT PRIMARY KEY,
    passenger_id INT NOT NULL,
    bus_id INT NOT NULL,
    route_id INT NOT NULL,
    seat_number VARCHAR(10),
    booking_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- replaces booking_date, payment_date, created_at
    amount DECIMAL(10,2) NOT NULL,
    status ENUM('confirmed', 'pending', 'completed', 'cancelled') DEFAULT 'confirmed',
    ticket_number VARCHAR(50) UNIQUE,
    payment_method ENUM('cash', 'card', 'upi', 'netbanking') DEFAULT 'cash',
    payment_status ENUM('paid', 'pending', 'failed') DEFAULT 'paid',
    FOREIGN KEY (passenger_id) REFERENCES passengers(id),
    FOREIGN KEY (bus_id) REFERENCES buses(id),
    FOREIGN KEY (route_id) REFERENCES routes(id)
);

-- 8. Schedules Table
CREATE TABLE IF NOT EXISTS schedules (
    id INT AUTO_INCREMENT PRIMARY KEY,
    bus_id INT NOT NULL,
    route_id INT NOT NULL,
    departure_time DATETIME NOT NULL,
    arrival_time DATETIME NOT NULL,
    driver_id INT,
    staff_id INT, -- operator or conductor
    status ENUM('scheduled', 'delayed', 'completed', 'cancelled') DEFAULT 'scheduled',
    FOREIGN KEY (bus_id) REFERENCES buses(id),
    FOREIGN KEY (route_id) REFERENCES routes(id),
    FOREIGN KEY (driver_id) REFERENCES drivers(id),
    FOREIGN KEY (staff_id) REFERENCES staff(id)
);
